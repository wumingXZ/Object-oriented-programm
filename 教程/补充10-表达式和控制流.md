## 〇、C 语言回顾：表达式和控制流

```c
// C 语言中的表达式和语句
#include <stdio.h>

int main() {
    int a = 10, b = 3;

    // 算术运算
    int sum = a + b, diff = a - b;
    int prod = a * b, quot = a / b, rem = a % b;

    // 控制流
    if (a > 5) printf("a > 5\n");
    for (int i = 0; i < 3; i++) printf("%d ", i);
    printf("\n");

    // C 的限制：
    // 1. 没有 bool 类型（C99 之前）
    // 2. 没有范围 for 循环
    // 3. 没有 switch 的 fallthrough 标注（C++17 有 [[fallthrough]]）
    // 4. 类型转换只能用 C 风格强制转换，不安全
    return 0;
}
```

## 一、运算符深入

### 1.1 算术、关系和逻辑运算符

```cpp {cmd=true}
#include <iostream>
#include <cmath>
using namespace std;

// 前向声明：expensive_check 在 main 之后定义
bool expensive_check();

int main() {
    // 算术：+, -, *, /, %, - (一元负)
    int a = 5, b = 2;
    cout << "a / b = " << (a / b) << endl;   // 整数除法 → 2
    cout << "a % b = " << (a % b) << endl;   // 取模 → 1
    // 注意：% 只能用于整数类型
    // (-m) / n 和 (-m) % n 的结果依赖于实现（C++11 规定向 0 截断）

    // 逻辑与和逻辑或的短路求值
    int *p = nullptr;
    // 短路：p != nullptr 为 false，&& 右边不会执行
    if (p != nullptr && *p == 42) {
        cout << "不会执行到这里" << endl;
    }
    // 短路：左边为 true，|| 右边不会执行
    if (true || expensive_check()) {
        cout << "短路求值: expensive_check() 没有被调用!" << endl;
    }

    // 关系运算符：<, <=, >, >=, ==, !=
    cout << boolalpha;  // 打印 true/false 而非 1/0
    cout << "3 < 5:  " << (3 < 5)  << endl;
    cout << "3 == 5: " << (3 == 5) << endl;

    return 0;
}

bool expensive_check() {
    cout << "(expensive_check 被执行了! 不应该看到这条消息)" << endl;
    return true;
}
```

### 1.2 赋值运算符

```cpp {cmd=true}
#include <iostream>
using namespace std;

int main() {
    // 赋值运算符是右结合的
    int i, j;
    i = j = 42;   // j = 42, 然后 i = j
    cout << "i = " << i << ", j = " << j << endl;

    // 复合赋值运算符
    i += 5;   // i = i + 5
    i -= 3;   // i = i - 3
    i *= 2;   // i = i * 2
    cout << "i after compound assignments: " << i << endl;

    // ⚠️ 常见陷阱：赋值 vs 比较
    int x = 0;
    // if (x = 42)  // 把 42 赋给 x，条件永远为 true！
    //     cout << "这是 bug！" << endl;
    // 正确写法：
    if (x == 42) {
        cout << "正确使用 ==" << endl;
    }

    return 0;
}
```

### 1.3 自增/自减运算符

```cpp {cmd=true}
#include <iostream>
using namespace std;

int main() {
    int i = 0;

    // 前置 ++：先加后用
    cout << "++i = " << ++i << " (i = " << i << ")" << endl;  // i=1, 输出1

    // 后置 ++：先用后加
    cout << "i++ = " << i++ << " (i = " << i << ")" << endl;  // 输出1, i=2

    // ⚠️ 不要在同一个表达式里多次修改同一个变量
    // cout << i << " " << ++i << endl;  // 未定义行为！

    // 建议：自增/自减单独作为一条语句
    ++i;   // 清晰，无歧义

    return 0;
}
```

### 1.4 条件运算符 `?:`

```cpp {cmd=true}
#include <iostream>
#include <string>
using namespace std;

int main() {
    int grade = 75;

    // 条件运算符 (cond ? true_expr : false_expr)
    string result = (grade >= 60) ? "及格" : "不及格";
    cout << grade << " 分: " << result << endl;

    // 可以嵌套（但不要过度嵌套，影响可读性）
    string level = (grade >= 90) ? "优秀"
                 : (grade >= 80) ? "良好"
                 : (grade >= 60) ? "及格"
                 : "不及格";
    cout << "等级: " << level << endl;

    // 条件运算符的优先级很低，做输出时要加括号
    cout << "成绩: " << (grade < 60 ? "不及格" : "通过") << endl;

    return 0;
}
```

### 1.5 位运算符

```cpp {cmd=true}
#include <iostream>
#include <bitset>
using namespace std;

int main() {
    unsigned char bits = 0b00110101;  // C++14 二进制字面量

    cout << "原始:    " << bitset<8>(bits) << endl;
    cout << "~bits:   " << bitset<8>(~bits) << " (按位取反)" << endl;
    cout << "bits<<1: " << bitset<8>(bits << 1) << " (左移)" << endl;
    cout << "bits>>1: " << bitset<8>(bits >> 1) << " (右移)" << endl;

    unsigned char mask = 0b00001111;
    cout << "bits&mask: " << bitset<8>(bits & mask) << " (按位与)" << endl;
    cout << "bits|mask: " << bitset<8>(bits | mask) << " (按位或)" << endl;
    cout << "bits^mask: " << bitset<8>(bits ^ mask) << " (按位异或)" << endl;

    // 常见用法
    unsigned long quiz1 = 0;
    quiz1 |= 1UL << 27;     // 设置第27位
    quiz1 &= ~(1UL << 27);  // 清除第27位
    bool pass = quiz1 & (1UL << 27);  // 检查第27位

    cout << boolalpha << "第27位: " << pass << endl;

    return 0;
}
```

### 1.6 `sizeof` 运算符

```cpp {cmd=true}
#include <iostream>
#include <vector>
#include <string>
using namespace std;

int main() {
    cout << "=== sizeof 各种类型 ===" << endl;
    cout << "sizeof(char):     " << sizeof(char)     << endl;
    cout << "sizeof(int):      " << sizeof(int)      << endl;
    cout << "sizeof(int*):     " << sizeof(int*)     << endl;
    cout << "sizeof(string):   " << sizeof(string)   << endl;
    cout << "sizeof(vector<int>): " << sizeof(vector<int>) << endl;

    // sizeof 对数组和指针的区别
    int arr[10];
    int *p = arr;
    cout << "sizeof(arr): " << sizeof(arr) << " (整个数组)" << endl;
    cout << "sizeof(p):   " << sizeof(p)   << " (指针本身)" << endl;

    // 数组元素个数
    constexpr size_t sz = sizeof(arr) / sizeof(*arr);
    cout << "数组元素数: " << sz << endl;

    return 0;
}
```

### 1.7 类型转换

```cpp {cmd=true}
#include <iostream>
using namespace std;

int main() {
    // 隐式转换
    int ival = 3.541 + 3;  // 3.541 → 3, then 3+3=6
    cout << "ival = " << ival << endl;

    // C++ 强制类型转换（推荐！比 C 风格更安全、更明确）
    double d = 3.14;
    int i1 = static_cast<int>(d);     // 常规转换（编译期检查）
    cout << "static_cast: " << d << " → " << i1 << endl;

    // const_cast：去掉/加上 const
    const char *cs = "hello";
    char *s = const_cast<char*>(cs);  // 去掉 const（⚠️ 不要修改原数据！）

    // reinterpret_cast：位模式的重新解释（危险，少用）
    long addr = reinterpret_cast<long>(s);
    cout << "地址: " << addr << endl;

    // dynamic_cast 用于运行时类型检查（见补充17 - RTTI）

    return 0;
}
```

| 转换类型 | 用途 | 安全性 |
|---|---|---|
| `static_cast` | 常规类型转换（数值、void* 等） | 编译期检查 |
| `const_cast` | 去掉/添加 const/volatile | ⚠️ 注意不要修改原 const 对象 |
| `reinterpret_cast` | 位模式重新解释 | ❌ 非常危险，少用 |
| `dynamic_cast` | 多态类型的运行时转换 | 运行时检查（有开销） |
| C 风格 `(type)` | 以上所有 + 更多 | ❌ 不推荐——太宽泛，难排查 |

## 二、控制流语句深入

### 2.1 `switch` 语句

```cpp {cmd=true}
#include <iostream>
using namespace std;

int main() {
    // switch 语句：比多个 if-else 更清晰
    char grade = 'B';

    switch (grade) {
        case 'A':
            cout << "优秀！" << endl;
            break;   // ⚠️ 不要忘记 break！否则会 fall through
        case 'B':
            cout << "良好" << endl;
            break;
        case 'C':
            cout << "及格" << endl;
            break;
        case 'D':
            cout << "需要努力" << endl;
            break;
        default:
            cout << "无效的成绩" << endl;
            break;
    }

    // 故意的 fall through（多个 case 共享代码）
    char ch = 'a';
    unsigned vowelCnt = 0;
    switch (ch) {
        case 'a': case 'e': case 'i': case 'o': case 'u':
        case 'A': case 'E': case 'I': case 'O': case 'U':
            ++vowelCnt;
            break;
    }
    cout << "元音计数: " << vowelCnt << endl;

    // C++17: [[fallthrough]] 标注故意的 fall through
    // switch (x) {
    //     case 1:
    //         do_something();
    //         [[fallthrough]];  // 明确告诉编译器：这是故意的
    //     case 2:
    //         ...
    // }

    return 0;
}
```

> **switch 检查清单：**
> - `case` 标签必须是整数常量表达式
> - 每个 `case` 后面通常要有 `break`
> - 可以在最后一个 `case` 或 `default` 省略 `break`
> - 变量定义需要放在 `{ }` 块中

### 2.2 范围 `for` 循环（C++11）

```cpp {cmd=true}
#include <iostream>
#include <vector>
#include <string>
using namespace std;

int main() {
    vector<int> v{1, 2, 3, 4, 5};

    // 范围 for：读（不修改）
    cout << "读: ";
    for (auto x : v) {
        cout << x << " ";
    }
    cout << endl;

    // 范围 for：写（修改原容器中的元素）
    cout << "写 (翻倍): ";
    for (auto &x : v) {
        x *= 2;
    }
    for (auto x : v) {
        cout << x << " ";
    }
    cout << endl;

    // 对 string 使用范围 for
    string s = "Hello";
    cout << "字符: ";
    for (auto c : s) {
        cout << c << " ";
    }
    cout << endl;

    // ⚠️ 范围 for 中不能修改容器大小（不能 push_back / erase）
    // for (auto x : v) {
    //     v.push_back(x);  // ❌ 未定义行为！
    // }

    return 0;
}
```

### 2.3 `do while` 循环

```cpp {cmd=true}
#include <iostream>
using namespace std;

int main() {
    // do while：至少执行一次循环体
    string password;
    int attempts = 0;

    do {
        cout << "请输入密码 (第" << (attempts + 1) << "次尝试): " << flush;
        // 模拟输入
        password = (attempts == 0) ? "wrong" : "correct";
        cout << password << endl;
        ++attempts;
    } while (password != "correct" && attempts < 3);

    if (password == "correct")
        cout << "登录成功!" << endl;
    else
        cout << "密码错误次数过多!" << endl;

    return 0;
}
```

### 2.4 `break`、`continue` 和 `goto`

```cpp {cmd=true}
#include <iostream>
using namespace std;

int main() {
    // break：跳出最近的循环或 switch
    cout << "break 示例: ";
    for (int i = 0; i < 10; ++i) {
        if (i == 5) break;      // 提前退出循环
        cout << i << " ";
    }
    cout << endl;

    // continue：跳过本次迭代的剩余部分
    cout << "continue 示例: ";
    for (int i = 0; i < 10; ++i) {
        if (i % 2 == 0) continue; // 跳过偶数
        cout << i << " ";
    }
    cout << endl;

    // goto：无条件跳转（⚠️ 少用，只在跳出深层嵌套时有价值）
    int matrix[3][3] = {{1, 2, 3}, {4, 5, 6}, {7, 8, 9}};
    bool found = false;

    for (int i = 0; i < 3; ++i) {
        for (int j = 0; j < 3; ++j) {
            if (matrix[i][j] == 5) {
                found = true;
                goto found_it;  // 直接跳出双重循环
            }
        }
    }
found_it:
    cout << "找到 5? " << boolalpha << found << endl;

    return 0;
}
```

## 三、运算符优先级速查

| 优先级 | 运算符 | 结合性 |
|---|---|---|
| 1 (最高) | `::` | 左 |
| 2 | `.` `->` `()` `[]` `++`(后) `--`(后) | 左 |
| 3 | `!` `~` `++`(前) `--`(前) `+`(一元) `-`(一元) `*`(解引用) `&`(取地址) `sizeof` `new` `delete` | **右** |
| 4 | `.*` `->*` | 左 |
| 5 | `*` `/` `%` | 左 |
| 6 | `+` `-` | 左 |
| 7 | `<<` `>>` | 左 |
| 8 | `<` `<=` `>` `>=` | 左 |
| 9 | `==` `!=` | 左 |
| 10 | `&` (按位与) | 左 |
| 11 | `^` | 左 |
| 12 | `\|` | 左 |
| 13 | `&&` | 左 |
| 14 | `\|\|` | 左 |
| 15 | `?:` | **右** |
| 16 | `=` `+=` `-=` 等 | **右** |
| 17 | `,` | 左 |

> **记忆口诀：** "算关逻条赋逗"（算术 → 关系 → 逻辑 → 条件 → 赋值 → 逗号），大多数左结合，只有一元、条件和赋值是右结合。

## 四、补充总结

### 核心要点

| 知识点 | 关键结论 |
|---|---|
| 短路求值 | `&&` 和 `||` 左侧为 false/true 时跳过右侧 |
| `switch` | 别忘 `break`；`case` 标签必须是常量；用 `{}` 包局部变量 |
| 范围 `for` | `for (auto &x : container)` — 修改用引用，只读用值或 const 引用 |
| `do while` | 至少执行一次；条件后面要有 `;` |
| 类型转换 | 用 `static_cast` 代替 C 风格转换 |
| 优先级 | 不确定时加括号——代码的可读性比简洁更重要 |

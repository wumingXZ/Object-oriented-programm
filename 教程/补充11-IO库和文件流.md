## 〇、C 语言回顾：文件操作

```c
#include <stdio.h>

int main() {
    FILE *fp = fopen("data.txt", "w");
    if (fp == NULL) { /* 错误处理 */ return 1; }

    fprintf(fp, "Hello, file!\n");
    fclose(fp);

    fp = fopen("data.txt", "r");
    char buf[100];
    fgets(buf, sizeof(buf), fp);
    printf("读到: %s", buf);
    fclose(fp);
    return 0;
}

// C 的问题：
// 1. FILE* 是裸指针，容易忘记 fclose（资源泄漏）
// 2. 格式化字符串 printf/scanf 不是类型安全的
// 3. 错误处理依赖于全局 errno
// 4. 没有 string 类型，字符串操作繁琐
```

## 一、IO 类层次结构

C++ 的 IO 库基于继承体系：

```
ios_base
  └── ios
       ├── istream  (cin)
       │    └── ifstream
       ├── ostream  (cout, cerr, clog)
       │    └── ofstream
       └── iostream
            ├── fstream
            └── stringstream (istringstream, ostringstream)
```

| 头文件 | 核心类 | 用途 |
|---|---|---|
| `<iostream>` | `istream`, `ostream`, `iostream` | 控制台 IO |
| `<fstream>` | `ifstream`, `ofstream`, `fstream` | 文件 IO |
| `<sstream>` | `istringstream`, `ostringstream`, `stringstream` | 字符串 IO |

## 二、文件输入输出 `<fstream>`

### 2.1 写入文件

```cpp {cmd=true}
#include <iostream>
#include <fstream>
#include <string>
using namespace std;

int main() {
    // ofstream 用于写文件
    ofstream outFile("output.txt");

    if (!outFile) {
        cerr << "无法打开文件!" << endl;
        return 1;
    }

    // 向文件写入（用法和 cout 完全一样！）
    outFile << "=== 学生成绩单 ===" << endl;
    outFile << "姓名: 张三" << endl;
    outFile << "期中: 85, 期末: 90" << endl;
    outFile << "最终成绩: 88.5" << endl;

    outFile.close();  // 显式关闭（析构时也会自动关闭）

    cout << "文件写入成功: output.txt" << endl;
    return 0;
}
```

### 2.2 读取文件

```cpp {cmd=true}
#include <iostream>
#include <fstream>
#include <string>
using namespace std;

int main() {
    // 先确保有文件可读
    ofstream("sample.txt") << "第一行: Hello\n第二行: World\n第三行: C++";

    // ifstream 用于读文件
    ifstream inFile("sample.txt");

    if (!inFile) {
        cerr << "无法打开文件!" << endl;
        return 1;
    }

    // 方式1：逐行读取
    cout << "=== 逐行读取 ===" << endl;
    string line;
    while (getline(inFile, line)) {
        cout << line << endl;
    }

    // 方式2：逐词读取（重置文件指针）
    inFile.clear();             // 清除 EOF 状态
    inFile.seekg(0, ios::beg);  // 回到文件开头

    cout << "\n=== 逐词读取 ===" << endl;
    string word;
    while (inFile >> word) {
        cout << "[" << word << "] ";
    }
    cout << endl;

    inFile.close();
    return 0;
}
```

### 2.3 文件打开模式

```cpp {cmd=true}
#include <iostream>
#include <fstream>
using namespace std;

int main() {
    // 常用打开模式
    // ios::in      — 读模式（ifstream 默认）
    // ios::out     — 写模式（ofstream 默认）
    // ios::app     — 追加模式（每次写入都在文件末尾）
    // ios::ate     — 打开后定位到文件尾
    // ios::trunc   — 截断文件（ofstream 默认）
    // ios::binary  — 二进制模式

    // 写模式 + 截断（默认）
    ofstream f1("test.txt");
    f1 << "第一次写入\n";
    f1.close();

    // 追加模式
    ofstream f2("test.txt", ios::app);
    f2 << "追加的内容\n";
    f2.close();

    // 验证
    ifstream f3("test.txt");
    string line;
    cout << "文件内容:" << endl;
    while (getline(f3, line)) {
        cout << "  " << line << endl;
    }

    return 0;
}
```

### 2.4 实战：读取学生成绩文件

```cpp {cmd=true}
#include <iostream>
#include <fstream>
#include <vector>
#include <string>
#include <sstream>
using namespace std;

// 结构体：存储一行学生数据
struct Student {
    string name;
    double midterm;
    double final;
    vector<double> homework;
};

int main() {
    // 创建示例输入文件
    {
        ofstream f("students.txt");
        f << "Alice 85 90 95 88 92 90 87\n";
        f << "Bob 72 75 80 78 85\n";
        f << "Charlie 93 95 98 90 92\n";
    }

    ifstream in("students.txt");
    if (!in) {
        cerr << "无法打开文件!" << endl;
        return 1;
    }

    vector<Student> students;
    string line;

    while (getline(in, line)) {
        istringstream record(line);
        Student s;
        record >> s.name >> s.midterm >> s.final;

        // 读取所有作业成绩
        double score;
        while (record >> score) {
            s.homework.push_back(score);
        }

        students.push_back(s);
    }

    // 输出结果
    cout << "=== 读取的学生数据 ===" << endl;
    for (const auto &s : students) {
        cout << s.name << ": 期中=" << s.midterm
             << " 期末=" << s.final
             << " 作业数=" << s.homework.size()
             << endl;
    }

    return 0;
}
```

## 三、字符串流 `<sstream>`

### 3.1 `istringstream` — 从字符串读数据

```cpp {cmd=true}
#include <iostream>
#include <sstream>
#include <string>
#include <vector>
using namespace std;

int main() {
    string data = "Alice 85 90 95 88 92";

    // istringstream 把 string 当输入流用
    istringstream iss(data);

    string name;
    iss >> name;  // 读名字

    vector<int> scores;
    int score;
    while (iss >> score) {
        scores.push_back(score);
    }

    cout << "姓名: " << name << endl;
    cout << "成绩: ";
    for (auto s : scores) cout << s << " ";
    cout << endl;

    return 0;
}
```

### 3.2 `ostringstream` — 往字符串写数据

```cpp {cmd=true}
#include <iostream>
#include <sstream>
#include <iomanip>
using namespace std;

int main() {
    // ostringstream 把 string 当输出流用
    ostringstream oss;

    // 格式化输出到字符串
    oss << "=== 成绩报告 ===" << endl;
    oss << fixed << setprecision(2);
    oss << "平均分: " << 88.567 << endl;
    oss << "最高分: " << 98.0 << endl;
    oss << "最低分: " << 72.5 << endl;

    // 获取最终的字符串
    string report = oss.str();
    cout << report;

    // 通用转换函数（C++11）
    int i = 42;
    double d = 3.14;
    string si = to_string(i);
    string sd = to_string(d);
    cout << "数字转字符串: " << si << ", " << sd << endl;

    // 字符串转数字（C++11）
    int i2 = stoi("42");
    double d2 = stod("3.14");
    cout << "字符串转数字: " << i2 << ", " << d2 << endl;

    return 0;
}
```

### 3.3 综合示例：解析 CSV 格式

```cpp {cmd=true}
#include <iostream>
#include <sstream>
#include <string>
#include <vector>
using namespace std;

int main() {
    string csv_data = "张三,85,90,88\n李四,72,75,80\n王五,93,95,98";

    istringstream iss(csv_data);
    string line;

    cout << "=== CSV 解析结果 ===" << endl;
    while (getline(iss, line)) {
        istringstream lineStream(line);
        string field;
        vector<string> fields;

        // 以逗号为分隔符
        while (getline(lineStream, field, ',')) {
            fields.push_back(field);
        }

        // 输出
        cout << "姓名: " << fields[0]
             << ", 期中: " << fields[1]
             << ", 期末: " << fields[2]
             << ", 作业: " << fields[3]
             << endl;
    }

    return 0;
}
```

## 四、IO 状态管理

### 4.1 流的条件状态

```cpp {cmd=true}
#include <iostream>
#include <sstream>
#include <limits>
using namespace std;

int main() {
    // 模拟用户输入：把 cin 替换为 istringstream 来演示
    // 实际场景中，cin 的行为和这里完全一样
    istringstream input("abc\n42");  // 第一次输入无效，第二次有效
    int number;

    cout << "模拟输入: 第一次='abc'(无效), 第二次='42'(有效)" << endl;

    // 检查输入是否成功
    while (!(input >> number)) {
        cout << "输入无效! 重试中..." << endl;
        input.clear();  // 清除错误状态
        input.ignore(numeric_limits<streamsize>::max(), '\n');  // 丢弃输入缓冲
    }

    cout << "成功读取: " << number << endl;
    return 0;
}
```

### 4.2 流状态标志

| 状态 | 含义 | 检测方法 |
|---|---|---|
| `goodbit` | 无错误 | `cin.good()` |
| `eofbit` | 到达文件尾 | `cin.eof()` |
| `failbit` | IO 操作失败（可恢复） | `cin.fail()` |
| `badbit` | 流已崩溃（不可恢复） | `cin.bad()` |

```cpp {cmd=true}
#include <iostream>
#include <sstream>
using namespace std;

int main() {
    // 演示各种状态
    istringstream iss("42");

    int x;
    iss >> x;
    cout << "读 42 后: good=" << iss.good()
         << " eof=" << iss.eof()
         << " fail=" << iss.fail()
         << " bad=" << iss.bad() << endl;

    // 再读一次——应该失败（没有更多数据）
    iss >> x;
    cout << "再读:     good=" << iss.good()
         << " eof=" << iss.eof()
         << " fail=" << iss.fail()
         << " bad=" << iss.bad() << endl;

    // 流可以隐式转为 bool——方便做条件判断
    if (iss) {
        cout << "流状态良好" << endl;
    } else {
        cout << "流状态不佳" << endl;
    }

    return 0;
}
```

## 五、`flush` / `ends` / `unitbuf` — 缓冲区管理

```cpp {cmd=true}
#include <iostream>
using namespace std;

int main() {
    // endl：换行 + 刷新
    cout << "Hello" << endl;

    // flush：只刷新，不换行
    cout << "正在处理..." << flush;
    // ... 处理完成 ...
    cout << "完成!" << endl;  // endl 隐含 flush

    // ends：插入 \0 + 刷新（stringstream 场景使用）
    // cout << "hello" << ends;  // 不常用

    // unitbuf：每次输出后自动刷新
    cout << unitbuf;      // 开启自动刷新
    cout << "A" << "B" << "C";
    cout << nounitbuf;    // 恢复默认缓冲

    return 0;
}
```

## 六、补充总结

### 核心要点

| 知识点 | 关键结论 |
|---|---|
| 文件 IO | `ifstream`/`ofstream` 用法和 `cin`/`cout` 一样 |
| 字符串流 | `istringstream` 用来解析字符串，`ostringstream` 用来构建字符串 |
| 打开模式 | `ios::app` 追加，`ios::trunc` 截断，`ios::binary` 二进制 |
| 状态检查 | 流可隐式转 `bool`——直接 `if (cin)` 用于判断 |
| `endl` vs `\n` | `endl` = `\n` + `flush`；不需要立即刷新时用 `\n` 性能更好 |
| 数字转换 | C++11: `to_string()` / `stoi()` / `stod()` |

### 与课程的联系

- **第03周** 讲的容器操作可以结合文件 IO 做数据持久化
- **第05周** 的类可以重载 `>>` 和 `<<` 操作符（见第07周），直接读写自定义类型
- 字符串流是实现"字符串到对象"的反序列化的核心工具

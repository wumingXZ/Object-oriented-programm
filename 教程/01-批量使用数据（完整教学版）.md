## 一、C++ 的第一个程序：Hello World

```cpp {cmd=true}
// 你的第一个 C++ 程序
#include <iostream>

int main() {
    std::cout << "Hello, world!" << std::endl;
    return 0;  // 返回 0 表示程序运行成功
}
```

### 逐行解析

| 代码元素 | 含义 |
|---|---|
| `#include <iostream>` | 包含标准输入输出流库 |
| `int main()` | 主函数——每个 C++ 程序必须有，是程序的入口 |
| `std::cout` | 标准输出流对象 |
| `<<` | 输出操作符（左结合） |
| `std::endl` | 结束当前输出行并刷新缓冲区 |
| `return 0;` | 向操作系统返回状态码 |

### 操作符 `<<` 的链式调用

`<<` 是**左结合**的，所以以下两行等价：

```cpp
std::cout << "Hello, " << "world!" << std::endl;
(std::cout << "Hello, ") << "world!" << std::endl;
// 每一次 << 都返回左边的 ostream 对象，因此可以继续链式调用
```

## 二、名字空间 `std::`

```cpp {cmd=true}
#include <iostream>

// 方式1：使用 using 声明（推荐，精确控制）
using std::cout;
using std::endl;

// 方式2：直接使用全限定名
// 方式3：using namespace std;（不推荐在头文件中使用）

int main() {
    cout << "无需 std:: 前缀了!" << endl;
    return 0;
}
```

- `std::` 表明名字属于名字空间 `std`（标准库的名字空间）
- `::` 是**作用域操作符**
- 在 `.cpp` 文件中使用 `using namespace std;` 是常见的教学写法

## 三、字符串类型 `std::string`

### 3.1 基本使用：读入名字并打招呼

```cpp {cmd=true}
#include <iostream>
#include <string>
using namespace std;

int main() {
    cout << "请输入你的名字: ";
    string name;         // 定义一个 string 变量
    cin >> name;         // 从标准输入读入（遇到空白停止）
    cout << "Hello, " << name << "!" << endl;
    return 0;
}
```

### 3.2 字符串的连接

```cpp {cmd=true}
#include <iostream>
#include <string>
using namespace std;

int main() {
    string greeting = "Hello, " + string("world") + "!";
    cout << greeting << endl;
    cout << "字符串长度: " << greeting.size() << endl;
    return 0;
}
```

### 3.3 装框输出 — 综合练习

```cpp {cmd=true}
#include <iostream>
#include <string>
using namespace std;

int main() {
    cout << "请输入你的名字: ";
    string name;
    cin >> name;

    // 构建问候语
    const string greeting = "Hello, " + name + "!";

    // 构建框框
    const string spaces(greeting.size(), ' ');
    const string second = "* " + spaces + " *";
    const string first(second.size(), '*');

    // 输出
    cout << endl;
    cout << first << endl;
    cout << second << endl;
    cout << "* " << greeting << " *" << endl;
    cout << second << endl;
    cout << first << endl;

    return 0;
}
```

### 3.4 `string` 常用操作汇总

| 操作 | 说明 |
|---|---|
| `string s` | 定义空字符串 |
| `string s = "hello"` | 用 C 字符串初始化 |
| `string s(n, c)` | 用 n 个字符 c 初始化 |
| `os << s` | 输出 |
| `is >> s` | 读入（跳过空白，读到下一个空白停止） |
| `s + t` | 连接两个字符串 |
| `s.size()` | 返回字符个数 |
| `s[i]` | 访问第 i 个字符（从 0 开始） |

`★ Insight ─────────────────────────────────────`
`const string spaces(greeting.size(), ' ');` 使用了**构造函数**来创建对象。`string(count, char)` 构造包含 `count` 个 `char` 的字符串。C++ 标准库中的类型通过多种构造函数提供了灵活的初始化方式——这被称为**构造函数重载**。

## 四、批量数据处理 — `vector` 容器

### 4.1 问题：计算学生成绩

场景：期末考试成绩占 40%，期中 20%，**作业中值**占 40%。需要收集所有作业成绩，排序后取中值。

```cpp {cmd=true}
#include <iostream>
#include <iomanip>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    // 模拟输入数据
    cout << "=== 学生成绩计算 ===" << endl;

    string name = "Alice";
    double midterm = 85, final = 90;

    // vector：动态数组，自动管理大小
    vector<double> homework;

    // 模拟读入作业成绩
    double grades[] = {95, 88, 92, 90, 87, 93};
    for (double g : grades) {
        homework.push_back(g);  // 向 vector 尾部添加元素
    }

    cout << "读入了 " << homework.size() << " 个作业成绩: ";
    for (double g : homework) cout << g << " ";
    cout << endl;

    // 检查是否为空
    if (homework.empty()) {
        cout << "错误：没有作业成绩！" << endl;
        return 1;
    }

    // 排序
    sort(homework.begin(), homework.end());
    cout << "排序后: ";
    for (double g : homework) cout << g << " ";
    cout << endl;

    // 计算中值
    typedef vector<double>::size_type vec_sz;
    vec_sz size = homework.size();
    vec_sz mid = size / 2;
    double median = (size % 2 == 0)
        ? (homework[mid] + homework[mid-1]) / 2
        : homework[mid];

    cout << "中值: " << median << endl;

    // 计算最终成绩
    double final_grade = 0.2 * midterm + 0.4 * final + 0.4 * median;
    cout << fixed << setprecision(1);
    cout << "最终成绩: " << final_grade << endl;

    return 0;
}
```

### 4.2 流程控制：检测输入结束

```cpp {cmd=true}
#include <iostream>
#include <vector>
using namespace std;

int main() {
    cout << "请输入一些数字（用空格分隔），输入非数字结束:" << endl;

    vector<double> numbers;
    double x;
    while (cin >> x) {     // 读取成功 → 循环继续
        numbers.push_back(x);
    }                      // 读取失败（EOF 或非法输入）→ 退出

    // cin.clear() 可以恢复流状态，但这里跳过

    cout << "共读取了 " << numbers.size() << " 个数字" << endl;
    double sum = 0;
    for (double n : numbers) sum += n;
    if (!numbers.empty())
        cout << "平均值: " << sum / numbers.size() << endl;

    return 0;
}
```

> EOF（文件结束）触发方式：**Windows: Ctrl+Z**，**Mac/Linux: Ctrl+D**

### 4.3 `vector` 操作汇总

| 操作 | 说明 |
|---|---|
| `vector<T> v` | 创建空的 `vector` |
| `v.push_back(e)` | 在末尾添加元素 e |
| `v.size()` | 返回元素个数 |
| `v.empty()` | 检查是否为空 |
| `v[i]` | 访问第 i 个元素（不检查越界） |
| `v.begin()` | 返回指向第一个元素的迭代器 |
| `v.end()` | 返回指向最后一个元素**之后**的迭代器 |

### 4.4 输出精度控制

```cpp {cmd=true}
#include <iostream>
#include <iomanip>
using namespace std;

int main() {
    double pi = 3.14159265358979;

    // 默认：有效数字
    cout << "默认: " << pi << endl;
    cout << "setprecision(3): " << setprecision(3) << pi << endl;
    cout << "setprecision(10): " << setprecision(10) << pi << endl;

    // 固定小数位数
    cout << fixed;
    cout << "fixed + setprecision(2): " << setprecision(2) << pi << endl;
    cout << "fixed + setprecision(6): " << setprecision(6) << pi << endl;

    // 保存和恢复精度
    streamsize prec = cout.precision();  // 获取当前精度
    cout << setprecision(2) << "短格式: " << pi << endl;
    cout << setprecision(prec) << "恢复后: " << pi << endl;

    return 0;
}
```

| 操作 | 说明 |
|---|---|
| `setprecision(n)` | 设置输出精度为 n |
| `cout.precision()` | 返回当前精度值 |
| `streamsize` | `setprecision` 所期望值的类型 |
| `fixed` | 切换为定点小数模式（控制小数位数而非有效数字） |

## 五、完整的成绩计算程序

```cpp {cmd=true}
#include <iostream>
#include <iomanip>
#include <string>
#include <vector>
#include <algorithm>
#include <sstream>
using namespace std;

// 模拟 cin 的输入（实际运行时用 cin 代替 iss）
// 格式：姓名 期中成绩 期末成绩 作业成绩1 作业成绩2 ... EOF
const char* SIMULATED_INPUT = "Alice 85 90 95 88 92 90 87 93";

int main() {
    // 用字符串流模拟输入（实际使用中替换为 cin）
    istringstream iss(SIMULATED_INPUT);
    // 以下代码与直接使用 cin 完全一样，只需将所有 cin 替换为 iss

    // 输入姓名
    cout << "请输入你的名字: ";
    string name;
    iss >> name;  // 实际使用: cin >> name;
    cout << "Hello, " << name << "!" << endl;

    // 输入期中、期末成绩
    cout << "请输入期中、期末成绩: ";
    double midterm, final;
    iss >> midterm >> final;  // 实际使用: cin >> midterm >> final;
    cout << "期中: " << midterm << ", 期末: " << final << endl;

    // 输入作业成绩
    cout << "请输入所有作业成绩 (EOF 结束):" << endl;
    vector<double> homework;
    double x;
    while (iss >> x)  // 实际使用: while (cin >> x)
        homework.push_back(x);
    cout << "共读取 " << homework.size() << " 个作业成绩: ";
    for (double g : homework) cout << g << " ";
    cout << endl;

    // 检查是否有作业成绩
    typedef vector<double>::size_type vec_sz;
    vec_sz size = homework.size();
    if (size == 0) {
        cout << "错误：没有录入作业成绩！" << endl;
        return 1;
    }

    // 排序并计算中值
    sort(homework.begin(), homework.end());
    cout << "排序后: ";
    for (double g : homework) cout << g << " ";
    cout << endl;

    vec_sz mid = size / 2;
    double median = (size % 2 == 0)
        ? (homework[mid] + homework[mid-1]) / 2
        : homework[mid];
    cout << "中值: " << median << endl;

    // 计算并输出最终成绩（期中20% + 期末40% + 作业中值40%）
    streamsize prec = cout.precision();
    cout << "最终成绩: " << setprecision(3)
         << 0.2 * midterm + 0.4 * final + 0.4 * median
         << setprecision(prec) << endl;

    // ★ C 语言对比 ★
    // 在 C 语言中，你需要手动管理数组大小、用 qsort 排序、自己写中值逻辑
    // C++ 的 vector 自动管理内存，sort 函数一行搞定排序
    cout << "\n--- C vs C++ 对比 ---" << endl;
    cout << "C:  需要 int arr[100]; 预设大小，可能越界或浪费" << endl;
    cout << "C++: vector<double> hw; 自动增长，不越界不浪费" << endl;

    return 0;
}
```

## 六、第一周总结

| 知识点 | 说明 |
|---|---|
| `#include <iostream>` | 标准输入输出 |
| `std::cout` / `std::cin` | 输出流 / 输入流 |
| `<<` / `>>` | 输出操作符 / 输入操作符 |
| `std::string` | 字符串类型：`+` 连接，`.size()` 长度，`[i]` 索引 |
| `std::vector<T>` | 动态数组：`.push_back()` 添加，`.size()` 大小 |
| `std::sort(b, e)` | 对 `[b, e)` 区间按非降序排列 |
| `setprecision(n)` | 设置输出精度 |
| `while(cin >> x)` | 边读边检测输入结束 |

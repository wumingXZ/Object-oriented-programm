<a id="一命名空间-namespace"></a>
## 一、命名空间 `namespace`

### 1.1 为什么需要命名空间？

```cpp {cmd=true}
#include <iostream>
using namespace std;

// 问题：不同库定义了同名的类和函数
namespace Audio {
    class File {
    public:
        void open() { cout << "打开音频文件..." << endl; }
    };
}

namespace Video {
    class File {
    public:
        void open() { cout << "打开视频文件..." << endl; }
    };
}

int main() {
    // 通过命名空间区分同名的类
    Audio::File audioFile;
    Video::File videoFile;

    audioFile.open();
    videoFile.open();

    return 0;
}
```

### 1.2 命名空间的基本用法

```cpp {cmd=true}
#include <iostream>

// 定义命名空间
namespace MyLib {
    int version = 1;
    void greet() { std::cout << "Hello from MyLib v" << version << std::endl; }

    // 命名空间可以嵌套
    namespace Detail {
        void helper() { std::cout << "内部辅助函数" << std::endl; }
    }
}

// 命名空间可以分开定义（同一个命名空间可以在多处打开）
namespace MyLib {
    void farewell() { std::cout << "Goodbye from MyLib" << std::endl; }
}

int main() {
    // using 声明：引入单个名字
    using MyLib::greet;
    greet();

    // 完整限定名访问
    MyLib::farewell();
    MyLib::Detail::helper();

    // 命名空间别名
    namespace MLD = MyLib::Detail;
    MLD::helper();

    return 0;
}
```

### 1.3 `using` 声明 vs `using` 指示

```cpp {cmd=true}
#include <iostream>

// using 声明：引入单个名字（推荐）
using std::cout;
using std::endl;

// ❌ using 指示：引入整个命名空间（不推荐在头文件）
// using namespace std;  // 污染全局命名空间！

int main() {
    cout << "using 声明：精确控制，避免名字冲突" << endl;
    return 0;
}
```

| | `using` 声明 | `using` 指示 |
|---|---|---|
| 语法 | `using std::cout;` | `using namespace std;` |
| 作用 | 引入单个名字 | 引入整个命名空间 |
| 冲突风险 | 低（显式指定） | 高（大量名字涌入） |
| 推荐场景 | `.cpp` 文件和 `.h` 文件均可 | **仅限 `.cpp` 文件** |

> **铁律：永远不要在头文件中使用 `using namespace std;` 或 `using` 指示！** 这会让所有包含该头文件的 `.cpp` 文件都被强制污染。

### 1.4 无名命名空间 — 文件私有的替代品

```cpp {cmd=true}
#include <iostream>

// 无名命名空间：名字仅在当前文件可见（取代 C 的 static）
namespace {
    int internal_counter = 0;  // 仅在当前翻译单元可见

    void internal_func() {
        std::cout << "内部函数, counter = " << internal_counter << std::endl;
    }
}

int main() {
    internal_func();
    internal_counter++;
    internal_func();
    return 0;
}
```

> C++ 推荐用无名命名空间代替 C 的 `static` 来定义文件私有的变量和函数。

## 二、异常处理进阶

### 2.1 标准异常类层次

C++ 标准库定义了一套异常类继承体系：

```
exception
├── logic_error
│   ├── invalid_argument
│   ├── out_of_range
│   ├── length_error
│   └── domain_error
├── runtime_error
│   ├── overflow_error
│   ├── underflow_error
│   └── range_error
└── bad_alloc
    └── bad_cast (等)
```

```cpp {cmd=true}
#include <iostream>
#include <stdexcept>
#include <vector>
#include <string>
using namespace std;

// 不安全的除法和越界检查
int safe_divide(int a, int b) {
    if (b == 0)
        throw invalid_argument("除数不能为零!");
    return a / b;
}

int safe_get(const vector<int> &v, size_t idx) {
    if (idx >= v.size())
        throw out_of_range("索引 " + to_string(idx)
                         + " 越界，vector 大小为 " + to_string(v.size()));
    return v[idx];
}

int main() {
    // 按异常类型分别捕获
    try {
        cout << safe_divide(10, 2) << endl;
        // cout << safe_divide(10, 0) << endl;  // 抛出 invalid_argument
    } catch (const invalid_argument &e) {
        cerr << "参数错误: " << e.what() << endl;
    } catch (const out_of_range &e) {
        cerr << "越界错误: " << e.what() << endl;
    } catch (const exception &e) {
        cerr << "其他异常: " << e.what() << endl;
    }

    // 越界捕获
    vector<int> v{1, 2, 3};
    try {
        cout << safe_get(v, 5) << endl;
    } catch (const out_of_range &e) {
        cerr << "捕获到: " << e.what() << endl;
    }

    return 0;
}
```

### 2.2 `catch` 的匹配规则

```cpp {cmd=true}
#include <iostream>
#include <stdexcept>
using namespace std;

int main() {
    try {
        throw runtime_error("一个运行时错误");
    }
    // catch 块从上到下匹配，找到第一个兼容的就停止
    catch (const runtime_error &e) {     // 先匹配更具体的
        cerr << "[runtime_error] " << e.what() << endl;
    }
    catch (const exception &e) {         // 后匹配更宽泛的
        cerr << "[exception] " << e.what() << endl;
    }
    // catch (...) { ... }               // 捕获所有异常（放在最后）

    // ⚠️ 注意：catch 的顺序很重要！派生类要放在基类前面
    // catch (const exception &e) 在前 → runtime_error 永远不会被后面的 catch 捕获

    return 0;
}
```

### 2.3 构造函数中的 `try` 块

```cpp {cmd=true}
#include <iostream>
#include <stdexcept>
#include <string>
using namespace std;

class Resource {
    string name_;
    int *data_;
public:
    // 构造函数 try 块：捕获初始化列表中的异常
    Resource(const string &name, int size)
        try : name_(name), data_(new int[size]) {
            if (size <= 0)
                throw invalid_argument("size 必须 > 0");
            cout << "资源 " << name_ << " 创建成功, 大小=" << size << endl;
        } catch (...) {
            // 初始化列表中的异常会在这里被捕获
            cerr << "构造 " << name_ << " 失败!" << endl;
            throw;  // 构造函数 try 块会自动重新抛出
        }

    ~Resource() {
        delete[] data_;
        cout << "资源 " << name_ << " 释放" << endl;
    }
};

int main() {
    try {
        Resource r1("R1", 100);
        Resource r2("R2", -5);   // 抛出异常
    } catch (const exception &e) {
        cerr << "main 捕获: " << e.what() << endl;
    }
    return 0;
}
```

### 2.4 `noexcept` — 承诺不抛异常

```cpp {cmd=true}
#include <iostream>
#include <vector>
#include <string>
using namespace std;

// noexcept：承诺不抛异常（编译器可以优化，标准库容器会检查）
void safe_func() noexcept {
    cout << "这个函数承诺不抛异常" << endl;
}

void might_throw() {
    cout << "这个函数可能抛异常" << endl;
}

// noexcept 可以作为运算符检查表达式
template <typename T>
void check_noexcept(T &&func, const string &name) {
    cout << name << " noexcept? "
         << boolalpha << noexcept(func()) << endl;
}

int main() {
    safe_func();

    // 编译期检查
    cout << "safe_func noexcept? " << noexcept(safe_func()) << endl;
    cout << "might_throw noexcept? " << noexcept(might_throw()) << endl;

    // 移动操作应该标记 noexcept（否则 vector 扩容会用拷贝）
    cout << "string 移动构造 noexcept? "
         << noexcept(string(move(*(new string())))) << endl;

    // 条件 noexcept
    auto lambda = []() noexcept(true) { /* 不抛异常 */ };
    cout << "lambda noexcept? " << noexcept(lambda()) << endl;

    return 0;
}
```

## 三、自定义异常类

```cpp {cmd=true}
#include <iostream>
#include <stdexcept>
#include <string>
using namespace std;

// 自定义异常：继承自标准异常类
class DatabaseException : public runtime_error {
public:
    explicit DatabaseException(const string &msg)
        : runtime_error("数据库错误: " + msg) {}
};

class ConnectionException : public DatabaseException {
    string host_;
public:
    ConnectionException(const string &host, const string &msg)
        : DatabaseException(msg), host_(host) {}

    const string &host() const { return host_; }
};

int main() {
    try {
        throw ConnectionException("db.example.com", "连接超时");
    } catch (const ConnectionException &e) {
        cerr << e.what() << endl;
        cerr << "主机: " << e.host() << endl;
    } catch (const DatabaseException &e) {
        cerr << e.what() << endl;
    } catch (const exception &e) {
        cerr << "未知错误: " << e.what() << endl;
    }

    return 0;
}
```

## 四、补充总结

### 命名空间速查

| 特性 | 语法 | 用途 |
|---|---|---|
| 定义 | `namespace N { ... }` | 创建命名空间 |
| 使用单个名字 | `using N::f;` | 精确引入 |
| 使用整个空间 | `using namespace N;` | 批量引入（仅 .cpp 文件） |
| 别名 | `namespace N2 = N1;` | 简化长名字 |
| 嵌套 | `namespace A::B { ... }` | 层次化（C++17: `namespace A::B { }`) |
| 匿名 | `namespace { ... }` | 文件私有 |

### 异常处理速查

| 知识点 | 关键结论 |
|---|---|
| 捕获顺序 | 派生类在前，基类在后 |
| `catch (...)` | 捕获所有异常，放在最后 |
| `exception::what()` | 获取错误信息 |
| 构造函数 try | 捕获初始化列表中的异常 |
| `noexcept` | 承诺不抛异常；移动操作必须标记 |
| 自定义异常 | 继承自 `exception` / `runtime_error` / `logic_error` |
| 不抛就析构 | 永远不要从析构函数中抛出异常！ |
